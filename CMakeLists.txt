cmake_minimum_required(VERSION 3.20)
project(MachDefense LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(MachDefenseCore STATIC Sources/Core/SaveData.cpp Sources/Core/DataTable.cpp
    Sources/Core/Chunk.cpp Sources/Core/Battle.cpp Sources/Core/Combat.cpp Sources/Core/Economy.cpp
    Sources/Core/WeaponController.cpp Sources/Core/Menu.cpp Sources/Core/Production.cpp
    Sources/Core/Bipod.cpp Sources/Core/Projectile.cpp Sources/Core/Transition.cpp Sources/Core/PlayerParts.cpp
    Sources/Core/Settlement.cpp Sources/Core/GameHUD.cpp Sources/Core/World.cpp
    Sources/Core/UpgradeDialog.cpp Sources/Core/SummaryDialog.cpp Sources/Core/GameSession.cpp Sources/Core/Loot.cpp Sources/Core/SpecialAttack.cpp Sources/Core/Formation.cpp Sources/Core/ValidationPanel.cpp Sources/Core/ValidationTools.cpp Sources/Core/Trail.cpp Sources/Core/WorldEffects.cpp)
target_include_directories(MachDefenseCore PUBLIC Sources/Core)
target_compile_options(MachDefenseCore PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off -fno-fast-math)
include(CTest)
if(BUILD_TESTING)
    add_executable(SaveDataTests Tests/SaveDataTests.cpp)
    target_link_libraries(SaveDataTests PRIVATE MachDefenseCore)
    target_compile_options(SaveDataTests PRIVATE -UNDEBUG)
    add_test(NAME SaveData COMMAND SaveDataTests)
    foreach(component ActionAudio DataTable Battle Button Combat Economy WeaponController Menu Production Bipod Projectile Transition PlayerParts Settlement GameHUD World UpgradeDialog SummaryDialog Loot GameSession SpecialAttack WorldSpecial Formation ValidationPanel ValidationTools Campaign CoreContent WorldEffects NaturalCampaign Trail SaveFailure FrameIO)

        target_link_libraries(${component}Tests PRIVATE MachDefenseCore)
        target_compile_options(${component}Tests PRIVATE -UNDEBUG -ffp-contract=off -fno-fast-math)
    endforeach()

    add_test(NAME DataTable COMMAND DataTableTests ${CMAKE_CURRENT_SOURCE_DIR}/Resources/Tables)

    add_test(NAME Combat COMMAND CombatTests ${CMAKE_CURRENT_SOURCE_DIR}/Resources/Tables)
    add_test(NAME Economy COMMAND EconomyTests ${CMAKE_CURRENT_SOURCE_DIR}/Resources/Tables)
    add_test(NAME WeaponController COMMAND WeaponControllerTests)
    add_test(NAME Menu COMMAND MenuTests ${CMAKE_CURRENT_SOURCE_DIR}/Resources)
    add_test(NAME Production COMMAND ProductionTests ${CMAKE_CURRENT_SOURCE_DIR}/Resources/Tables)
    add_test(NAME Bipod COMMAND BipodTests)
    add_test(NAME Projectile COMMAND ProjectileTests)
    add_test(NAME Transition COMMAND TransitionTests ${CMAKE_CURRENT_SOURCE_DIR}/Resources)
    add_test(NAME PlayerParts COMMAND PlayerPartsTests ${CMAKE_CURRENT_SOURCE_DIR}/Resources)
